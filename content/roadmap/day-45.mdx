## Chunk, embed, index

```python
import numpy as np
import re
from sentence_transformers import SentenceTransformer

model = SentenceTransformer("all-MiniLM-L6-v2")

def chunk(text: str, size: int = 800, overlap: int = 100) -> list[str]:
    step = size - overlap
    return [text[i:i + size] for i in range(0, len(text), step)]

class Index:
    def __init__(self, chunks: list[str]):
        self.chunks = chunks
        E = model.encode(chunks)
        self.M = E / np.linalg.norm(E, axis=1, keepdims=True)  # unit vectors

    def search(self, query: str, k: int = 3):
        q = model.encode([query])
        q = q / np.linalg.norm(q, axis=1, keepdims=True)
        scores = (self.M @ q.T).ravel()          # brute force: fine at this scale
        top = np.argsort(-scores)[:k]
        return [(self.chunks[i], float(scores[i]), i) for i in top]
```

## Retrieve, generate, cite, refuse

```python
PROMPT = """Answer using ONLY the numbered sources. Cite them as [n].
If the sources do not answer the question, reply exactly: NOT_IN_SOURCES.

Sources:
{sources}

Question: {question}
"""

def answer(question: str, index: Index, llm, k: int = 3):
    hits = index.search(question, k)
    sources = "\n\n".join(f"[{i}] {c}" for i, (c, s, _) in enumerate(hits))
    out = llm(PROMPT.format(sources=sources, question=question), temperature=0.0)
    return out, hits
```

## The 10-question eval: hits@k and citation accuracy

Two metrics, both from Day 38's definitions. hits@k asks whether retrieval
found the gold chunk; citation accuracy asks whether the answer actually
cited it:

```python
def evaluate(questions, index: Index, llm, k: int = 3):
    hits_k, cite_acc = [], []
    for q, gold in questions:
        out, hits = answer(q, index, llm, k)
        top_ids = [h[2] for h in hits]                    # global chunk ids
        hits_k.append(gold in top_ids)
        cited = {top_ids[int(n)] for n in re.findall(r"\[(\d+)\]", out)
                 if int(n) < len(top_ids)}
        cite_acc.append(bool(cited) and gold in cited)
    n = len(questions)
    return sum(hits_k) / n, sum(cite_acc) / n
```
